const BRIDGE_VERSION = 2;
const TIMEZONE = "Asia/Kuala_Lumpur";

function doGet(e) {
  const params = (e && e.parameter) ? e.parameter : {};
  const action = String(params.action || "").trim();
  const plot = sanitizePlotName(params.plot || "Plot1");
  const callback = params.callback ? String(params.callback).replace(/[^a-zA-Z0-9_$.]/g, "") : "";

  try {
    if (action === "setConfig") {
      const updated = savePlotConfig(plot, params);
      syncConfigSheetRow(plot, updated);
      return respond({ ok: true, v: BRIDGE_VERSION, action: "setConfig", plot: plot, config: updated }, callback);
    }

    if (action === "getConfig") {
      const cfg = loadPlotConfig(plot);
      if (params.fw || params.rssi || params.ackTareId || params.applied) {
        recordEsp32CheckIn(plot, cfg, params, false);
      }
      return respond(buildEspResponse(plot, cfg, false), callback);
    }

    if (params.weight !== undefined && String(params.weight).trim() !== "") {
      const weightVal = parseFloat(params.weight);
      if (isNaN(weightVal)) return respond({ ok: false, v: BRIDGE_VERSION, error: "Invalid weight" }, callback);
      appendWeightReading(plot, weightVal);
      const cfg = loadPlotConfig(plot);
      recordEsp32CheckIn(plot, cfg, params, true);
      return respond(buildEspResponse(plot, cfg, true), callback);
    }

    return respond({ ok: true, v: BRIDGE_VERSION, service: "KH CocoScale Cloud Bridge v2.0", plot: plot, config: loadPlotConfig(plot) }, callback);
  } catch (err) {
    return respond({ ok: false, v: BRIDGE_VERSION, error: String(err && err.message ? err.message : err) }, callback);
  }
}

function sanitizePlotName(raw) {
  const cleaned = String(raw || "Plot1").replace(/[^a-zA-Z0-9_-]/g, "");
  return cleaned || "Plot1";
}

function getDefaultConfig() {
  return { intervalMin: 1, powerMode: "always_on", lcdMode: "on", tareId: 0, updatedAt: "", lastSeenAt: "", lastAckTareId: 0, lastRssi: "", lastWeightKg: "", appliedState: "" };
}

function loadPlotConfig(plot) {
  const raw = PropertiesService.getScriptProperties().getProperty("CFG_" + plot);
  const base = getDefaultConfig();
  if (!raw) return base;
  try { return Object.assign(base, JSON.parse(raw)); } catch (_) { return base; }
}

function savePlotConfig(plot, params) {
  const cfg = loadPlotConfig(plot);
  if (params.intervalMin !== undefined) {
    const m = parseInt(params.intervalMin, 10);
    if (!isNaN(m)) cfg.intervalMin = Math.max(1, Math.min(240, m));
  }
  if (params.powerMode === "always_on" || params.powerMode === "smart_saver" || params.powerMode === "deep_sleep") cfg.powerMode = params.powerMode;
  if (["on", "text_only", "flash", "off"].indexOf(String(params.lcdMode)) >= 0) cfg.lcdMode = String(params.lcdMode);
  if (String(params.tareCmd || "") === "1") cfg.tareId = Math.floor(Date.now() / 1000);
  cfg.updatedAt = Utilities.formatDate(new Date(), TIMEZONE, "yyyy-MM-dd HH:mm:ss");
  PropertiesService.getScriptProperties().setProperty("CFG_" + plot, JSON.stringify(cfg));
  return cfg;
}

function recordEsp32CheckIn(plot, cfg, params, forceSheetUpdate) {
  const now = new Date();
  const prevSeen = cfg.lastSeenAt || "";
  let stateChanged = false;
  cfg.lastSeenAt = Utilities.formatDate(now, TIMEZONE, "yyyy-MM-dd HH:mm:ss");
  if (params.rssi !== undefined && params.rssi !== "") cfg.lastRssi = String(params.rssi);
  if (params.weight !== undefined && params.weight !== "") cfg.lastWeightKg = Number(parseFloat(params.weight).toFixed(3));
  if (params.ackTareId !== undefined && params.ackTareId !== "") {
    const ack = parseInt(params.ackTareId, 10);
    if (!isNaN(ack) && ack !== cfg.lastAckTareId) { cfg.lastAckTareId = ack; stateChanged = true; }
  }
  if (params.applied !== undefined && params.applied !== "" && cfg.appliedState !== String(params.applied)) {
    cfg.appliedState = String(params.applied);
    stateChanged = true;
  }
  const props = PropertiesService.getScriptProperties();
  props.setProperty("CFG_" + plot, JSON.stringify(cfg));
  const lastSw = Number(props.getProperty("LAST_SW_" + plot) || "0");
  if (forceSheetUpdate || stateChanged || !prevSeen || (now.getTime() - lastSw >= 55000)) {
    props.setProperty("LAST_SW_" + plot, String(now.getTime()));
    syncConfigSheetRow(plot, cfg);
  }
}

function buildEspResponse(plot, cfg, logged) {
  return { ok: true, v: BRIDGE_VERSION, logged: Boolean(logged), plot: plot, intervalMin: Number(cfg.intervalMin) || 1, powerMode: cfg.powerMode || "always_on", lcdMode: cfg.lcdMode || "on", tareId: Number(cfg.tareId) || 0 };
}

function appendWeightReading(plot, weightVal) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(plot);
  if (!sheet) {
    if (plot === "Plot1") sheet = ss.getSheets()[0];
    else { sheet = ss.insertSheet(plot); sheet.appendRow(["Date", "Time", "Weight_kg"]); sheet.getRange("A1:C1").setFontWeight("bold"); }
  }
  const now = new Date();
  sheet.appendRow([Utilities.formatDate(now, TIMEZONE, "yyyy-MM-dd"), Utilities.formatDate(now, TIMEZONE, "HH:mm:ss"), Number(weightVal.toFixed(3))]);
}

function syncConfigSheetRow(plot, cfg) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName("DeviceConfig");
  const headers = ["Plot_Key", "Interval_Min", "Power_Mode", "LCD_Mode", "Tare_ID", "Updated_At", "Last_ESP32_Seen", "Last_Ack_Tare_ID", "Last_RSSI", "Applied_State"];
  if (!sheet) { sheet = ss.insertSheet("DeviceConfig"); sheet.appendRow(headers); sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold"); }
  const data = sheet.getDataRange().getValues();
  let targetRow = -1;
  for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === plot) { targetRow = i + 1; break; } }
  const rowValues = [plot, cfg.intervalMin, cfg.powerMode, cfg.lcdMode, cfg.tareId || 0, cfg.updatedAt || "", cfg.lastSeenAt || "", cfg.lastAckTareId || 0, cfg.lastRssi || "", cfg.appliedState || ""];
  if (targetRow > 0) sheet.getRange(targetRow, 1, 1, rowValues.length).setValues([rowValues]);
  else sheet.appendRow(rowValues);
}

function respond(obj, callback) {
  const json = JSON.stringify(obj);
  if (callback) return ContentService.createTextOutput(callback + "(" + json + ");").setMimeType(ContentService.MimeType.JAVASCRIPT);
  return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON);
}
